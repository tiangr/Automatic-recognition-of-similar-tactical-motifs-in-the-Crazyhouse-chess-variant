from pathlib import Path
import json

ROOT = Path(__file__).resolve().parents[1]
AUG = ROOT / "data" / "derived" / "tactics_1k_aug.jsonl"

def pocket_str(p):
    if not p:
        return "-"
    return " ".join([f"{k}{v}" for k,v in sorted(p.items())])

def main():
    # load all tactics by id
    items = {}
    with AUG.open("r", encoding="utf-8") as f:
        for line in f:
            r = json.loads(line)
            _id = r.get("event_id") or (r.get("site") + "_" + str(r.get("ply")))
            items[_id] = r

    # paste your query + top10 ids here:
    ids = [
        "https://lichess.org/N7w3FJpI_5",
        "https://lichess.org/FowcxrOo_5",
        "https://lichess.org/SPYgej2h_7",
        "https://lichess.org/N7w3FJpI_6",
        "https://lichess.org/8VXCqA7J_7",
        "https://lichess.org/v8x54ggi_7",
        "https://lichess.org/K2LI9MNr_7",
        "https://lichess.org/K2LI9MNr_5",
        "https://lichess.org/IWAMD0KT_3",
        "https://lichess.org/jxpL5tIk_8",
        "https://lichess.org/iYNDP88f_7",
    ]

    for k, _id in enumerate(ids):
        r = items.get(_id)
        if not r:
            print("Missing:", _id)
            continue

        pv = r.get("pv_before") or r.get("pv_prev") or []
        bm = r.get("bestmove_before") or r.get("best_prev")
        played = r.get("played_move")
        d = r.get("delta")

        print("\n" + ("QUERY" if k==0 else f"HIT {k}"), _id)
        print("ply:", r.get("ply"), "delta:", d)
        print("played:", played, "best:", bm)
        print("PV:", " ".join(pv[:8]))
        print("PV has drop?:", any("@" in m for m in pv))
        print("pockets W:", pocket_str(r.get("pockets_white")), "| B:", pocket_str(r.get("pockets_black")))

if __name__ == "__main__":
    main()
